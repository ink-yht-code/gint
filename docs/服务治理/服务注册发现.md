# 服务注册发现

服务注册发现模块用于管理微服务的注册和发现，支持多种注册中心。

## 功能特性

- 支持多种注册中心（Consul、Nacos等）
- 服务注册和注销
- 服务健康检查
- 服务发现和负载均衡
- 与Context集成，自动传递追踪ID

## 使用示例

```go
package main

import (
    "github.com/ink-yht-code/gint/discovery"
    "github.com/ink-yht-code/gint/discovery/consul"
)

func main() {
    // 初始化Consul注册中心
    dc := consul.NewClient(consul.Config{
        Addr: "localhost:8500",
    })
    
    // 注册服务
    err := dc.Register("user-service", "192.168.1.100:8080", []string{"user", "auth"})
    if err != nil {
        // 处理错误
    }
    
    // 发现服务
    instances, err := dc.Discover("order-service")
    if err != nil {
        // 处理错误
    }
    
    // 使用服务实例
    for _, instance := range instances {
        // 调用服务
    }
}
```

## 健康检查

```go
// 注册健康检查
err := dc.RegisterWithHealthCheck("user-service", "192.168.1.100:8080", []string{"user", "auth"}, discovery.HealthCheckConfig{
    HTTP: "/health",
    Interval: 10 * time.Second,
    Timeout: 5 * time.Second,
})
if err != nil {
    // 处理错误
}
```

## 接口定义

```go
type DiscoveryClient interface {
    // Register 注册服务
    Register(serviceName, address string, tags []string) error
    
    // RegisterWithHealthCheck 注册服务并设置健康检查
    RegisterWithHealthCheck(serviceName, address string, tags []string, healthCheckConfig HealthCheckConfig) error
    
    // Deregister 注销服务
    Deregister(serviceName, address string) error
    
    // Discover 发现服务实例
    Discover(serviceName string) ([]Instance, error)
    
    // Watch 监听服务实例变化
    Watch(serviceName string) (<-chan []Instance, error)
}
```
