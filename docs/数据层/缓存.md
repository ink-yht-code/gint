# 缓存抽象层

缓存模块提供统一接口，支持 `memory` 与 `redis` 两种实现。

## 功能特性

- 统一接口：`Get/Set/Delete/Exists/Expire/TTL`
- 支持批量能力（`BatchCache`）与计数器能力（`Counter`）
- 支持类型安全封装：`TypedCache[T]`
- 支持默认全局缓存实例

## 基本使用

```go
import (
    "context"
    "time"

    "github.com/ink-yht-code/gint/cache"
    "github.com/ink-yht-code/gint/cache/memory"
)

func example(ctx context.Context) error {
    c := memory.New()
    cache.SetDefault(c)

    if err := cache.Set(ctx, "user:1", []byte("tom"), 10*time.Minute); err != nil {
        return err
    }
    _, err := cache.Get(ctx, "user:1")
    return err
}
```

## Redis 使用

```go
import (
    "github.com/ink-yht-code/gint/cache/redis"
    goredis "github.com/redis/go-redis/v9"
)

rdb := goredis.NewClient(&goredis.Options{Addr: "127.0.0.1:6379"})
c := redis.New(rdb, "app")
```

## TypedCache 示例

```go
type User struct {
    ID   string `json:"id"`
    Name string `json:"name"`
}

userCache := cache.NewTypedCache[User](c, "user:")
_ = userCache.Set(ctx, "1", User{ID: "1", Name: "tom"}, 5*time.Minute)
u, _ := userCache.Get(ctx, "1")
_ = u
```

## 兼容性说明

- 旧版本中的 `cache.New()` 已不适用，请使用 `memory.New()` 或 `redis.New()`
- 缓存接口统一使用 `context.Context` 作为第一参数

## 错误处理说明

- 若未设置默认缓存实例，调用 `cache.Get/Set/Delete` 会返回 `ErrCacheNotInitialized`
- 缓存 miss 场景建议使用 `errors.Is(err, cache.ErrCacheNotFound)` 判定

## 批量操作

```go
// 批量获取
keys := []string{"user:1", "user:2", "user:3"}
values, err := cache.BatchGet(ctx, keys)

// 批量删除
err := cache.BatchDelete(ctx, keys)
```

## 计数器

```go
// 递增
count, err := cache.Incr(ctx, "counter:visits")

// 递减
count, err := cache.Decr(ctx, "counter:visits")

// 增加指定值
count, err := cache.IncrBy(ctx, "counter:visits", 10)
```

## 完整示例

```go
package main

import (
    "context"
    "encoding/json"
    "errors"
    "time"

    "github.com/ink-yht-code/gint/cache"
    "github.com/ink-yht-code/gint/cache/memory"
    "github.com/ink-yht-code/gint/cache/redis"
    goredis "github.com/redis/go-redis/v9"
)

type User struct {
    ID   string `json:"id"`
    Name string `json:"name"`
    Age  int    `json:"age"`
}

func main() {
    ctx := context.Background()

    // 初始化内存缓存
    memCache := memory.New()
    cache.SetDefault(memCache)

    // 或使用 Redis 缓存
    // rdb := goredis.NewClient(&goredis.Options{Addr: "127.0.0.1:6379"})
    // redisCache := redis.New(rdb, "app:")
    // cache.SetDefault(redisCache)

    // 基本操作
    user := User{ID: "1", Name: "Alice", Age: 30}
    data, _ := json.Marshal(user)
    
    // 设置缓存
    cache.Set(ctx, "user:1", data, 10*time.Minute)
    
    // 获取缓存
    val, err := cache.Get(ctx, "user:1")
    if err != nil {
        if errors.Is(err, cache.ErrCacheNotFound) {
            // 缓存未命中
        }
    } else {
        var u User
        json.Unmarshal(val, &u)
    }

    // 类型安全缓存
    userCache := cache.NewTypedCache[User](memCache, "user:")
    userCache.Set(ctx, "1", user, 10*time.Minute)
    u, _ := userCache.Get(ctx, "1")
    _ = u

    // 检查存在
    exists, _ := cache.Exists(ctx, "user:1")
    
    // 删除缓存
    cache.Delete(ctx, "user:1")
    
    // 设置过期时间
    cache.Expire(ctx, "user:1", 5*time.Minute)
    
    // 获取TTL
    ttl, _ := cache.TTL(ctx, "user:1")
    _ = ttl
}
```

## 最佳实践

### 1. 缓存穿透防护

```go
func GetUser(ctx context.Context, id string) (*User, error) {
    // 先查缓存
    userCache := cache.NewTypedCache[User](cache.Default(), "user:")
    u, err := userCache.Get(ctx, id)
    if err == nil {
        return &u, nil
    }
    
    // 缓存未命中，查数据库
    user, err := db.GetUser(id)
    if err != nil {
        // 数据库查询失败，缓存空值防止穿透
        cache.Set(ctx, "user:"+id, []byte("null"), 1*time.Minute)
        return nil, err
    }
    
    if user == nil {
        // 用户不存在，缓存空值
        cache.Set(ctx, "user:"+id, []byte("null"), 5*time.Minute)
        return nil, nil
    }
    
    // 缓存用户数据
    userCache.Set(ctx, id, *user, 1*time.Hour)
    return user, nil
}
```

### 2. 缓存雪崩防护

```go
// 使用随机过期时间避免缓存同时失效
func SetUserCache(ctx context.Context, user *User) error {
    userCache := cache.NewTypedCache[User](cache.Default(), "user:")
    
    // 随机过期时间：30-60分钟
    ttl := time.Duration(30+rand.Intn(30)) * time.Minute
    return userCache.Set(ctx, user.ID, *user, ttl)
}
```

### 3. 缓存预热

```go
func WarmupCache(ctx context.Context) error {
    userCache := cache.NewTypedCache[User](cache.Default(), "user:")
    
    // 加载热点数据
    users, err := db.GetHotUsers()
    if err != nil {
        return err
    }
    
    for _, user := range users {
        userCache.Set(ctx, user.ID, user, 1*time.Hour)
    }
    
    return nil
}
```
