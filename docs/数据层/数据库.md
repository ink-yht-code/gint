# 数据库工具

数据库工具提供了连接池管理、查询构建器和事务支持。

## 功能特性

- 支持MySQL、PostgreSQL等主流数据库
- 连接池管理
- 事务支持
- 查询构建器
- 与Context集成，自动传递追踪ID

## 使用示例

```go
package main

import (
    "github.com/ink-yht-code/gint/db"
)

func main() {
    // 初始化数据库连接
    err := db.Init(db.Config{
        Driver:   "mysql",
        DSN:      "user:password@tcp(localhost:3306)/dbname?charset=utf8mb4&parseTime=True&loc=Local",
        MaxOpen:  10,
        MaxIdle:  5,
        MaxLifetime: 3600,
    })
    if err != nil {
        // 处理错误
    }
    
    // 查询数据
    var user struct {
        ID   int
        Name string
    }
    err = db.QueryRow("SELECT id, name FROM users WHERE id = ?", 1).Scan(&user.ID, &user.Name)
    if err != nil {
        // 处理错误
    }
    
    // 执行事务
    err = db.WithTx(func(tx *db.Tx) error {
        _, err := tx.Exec("INSERT INTO users (name) VALUES (?)", "Alice")
        if err != nil {
            return err
        }
        
        _, err = tx.Exec("INSERT INTO users (name) VALUES (?)", "Bob")
        return err
    })
    if err != nil {
        // 处理错误
    }
}
```

## 查询构建器

```go
// 使用查询构建器
users := []User{}
err := db.Select("id, name").
    From("users").
    Where("age > ?", 18).
    OrderBy("created_at DESC").
    Limit(10).
    Scan(&users)
if err != nil {
    // 处理错误
}
```

## 事务管理

```go
// 手动事务管理
tx, err := db.Begin()
if err != nil {
    // 处理错误
}
defer tx.Rollback()

_, err = tx.Exec("INSERT INTO users (name) VALUES (?)", "Alice")
if err != nil {
    // 处理错误
}

err = tx.Commit()
if err != nil {
    // 处理错误
}
```
